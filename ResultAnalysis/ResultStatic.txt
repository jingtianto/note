    // ============================================================
    // 全局状态
    // ============================================================
    let schemes = [];
    let currentSchemeId = null;
    let uploadedFiles = [];
    let statisticsResults = null;
    let independentLayout = 'horizontal';
    let keywordsExpanded = true;

    // ============================================================
    // 颜色识别（基于 ExcelJS 的 fgColor.indexed 映射）
    // ============================================================
    // 映射表：fgColor.indexed → 颜色名称
    const COLOR_INDEXED_MAP = {
        10: '红色',   // failed
        11: '绿色',   // passed
        35: '浅蓝色', // signal missing
        52: '橙色',   // trigger not occurred
        13: '黄色',   // warning（可选）
        9: 'blank'    // 无色
    };
    const COLOR_MEANING = {
        '红色': 'failed',
        '绿色': 'passed',
        '浅蓝色': 'signal missing',
        '橙色': 'trigger not occurred',
        '黄色': 'warning',
        'blank': 'blank',
        '其他': 'unknown'
    };

    // 从单元格获取颜色名称
    function getCellColorName(cell) {
        if (!cell || !cell.style || !cell.style.fill) return 'blank';
        const fill = cell.style.fill;
        let indexed = null;

        // 优先读取 fgColor.indexed
        if (fill.fgColor && fill.fgColor.indexed !== undefined) {
            indexed = fill.fgColor.indexed;
        }
        // 如果 fgColor 没有 indexed，尝试 bgColor（但通常为空）
        else if (fill.bgColor && fill.bgColor.indexed !== undefined) {
            indexed = fill.bgColor.indexed;
        }

        // 如果获取到 indexed，进行映射
        if (indexed !== null) {
            if (COLOR_INDEXED_MAP[indexed]) {
                return COLOR_INDEXED_MAP[indexed];
            }
            // 未知索引，记录日志并返回 blank
            console.warn('未知 fgColor.indexed:', indexed, '单元格值:', cell.value);
            return 'blank';
        }

        // 如果没有 indexed，检查是否有 rgb 或 argb（备用）
        let rgb = null;
        if (fill.fgColor && fill.fgColor.argb) {
            rgb = fill.fgColor.argb;
        } else if (fill.fgColor && fill.fgColor.rgb) {
            rgb = fill.fgColor.rgb;
        }
        if (rgb) {
            // 简单判断：如果是白色或 #FFFFFF，视为 blank
            const hex = rgb.replace('#', '').toLowerCase();
            if (hex === 'ffffff' || hex === 'ffffffff') return 'blank';
            // 否则返回 '其他'
            return '其他';
        }

        return 'blank';
    }

    function getColorMeaning(colorName) {
        return COLOR_MEANING[colorName] || colorName;
    }

    function getColorArgb(colorName) {
        const map = {
            '绿色': 'FF00FF00',
            '黄色': 'FFFFFF00',
            '橙色': 'FFFF9900',
            '红色': 'FFFF0000',
            '浅蓝色': 'FF99CCFF'
        };
        return map[colorName] || null;
    }

    function getColorCss(colorName) {
        const argb = getColorArgb(colorName);
        if (argb) return '#' + argb.substring(2);
        if (colorName === 'blank' || colorName === '空') return '#f1f5f9';
        if (colorName === '其他') return '#e2e8f0';
        return '#ffffff';
    }

    // ============================================================
    // 日志管理
    // ============================================================
    function log(message, type = 'info') {
        const area = document.getElementById('logArea');
        const entry = document.createElement('div');
        entry.className = 'log-' + type;
        entry.textContent = message;
        area.appendChild(entry);
        area.scrollTop = area.scrollHeight;
        if (type === 'error') console.error(message);
        else if (type === 'success') console.log(message);
        else console.info(message);
    }

    function clearLog() {
        document.getElementById('logArea').innerHTML = '';
    }

    // ============================================================
    // 方案管理（保持不变）
    // ============================================================
    function loadSchemes() {
        try {
            const data = localStorage.getItem('excelStatsSchemes');
            if (data) schemes = JSON.parse(data);
            else schemes = [];
        } catch(e) { schemes = []; }
        renderSchemeSelector();
        renderSchemeList();
        if (schemes.length > 0) {
            selectScheme(schemes[0].id);
        } else {
            document.getElementById('schemeEditor').style.display = 'none';
        }
        document.getElementById('schemeCount').textContent = schemes.length + ' 个方案';
    }

    function saveSchemes() {
        localStorage.setItem('excelStatsSchemes', JSON.stringify(schemes));
        renderSchemeSelector();
        renderSchemeList();
        document.getElementById('schemeCount').textContent = schemes.length + ' 个方案';
    }

    function renderSchemeSelector() {
        const sel = document.getElementById('schemeSelector');
        const current = sel.value;
        sel.innerHTML = '';
        schemes.forEach(s => {
            const opt = document.createElement('option');
            opt.value = s.id;
            opt.textContent = s.name;
            sel.appendChild(opt);
        });
        if (current && schemes.some(s => s.id === current)) sel.value = current;
        else if (schemes.length > 0) sel.value = schemes[0].id;
    }

    function renderSchemeList() {
        const container = document.getElementById('schemeList');
        container.innerHTML = '';
        schemes.forEach(s => {
            const div = document.createElement('span');
            div.className = 'scheme-item';
            div.innerHTML = `<span>${s.name}</span> <span class="badge">${s.keywords ? s.keywords.length : 0}</span> <span class="del" onclick="deleteScheme('${s.id}')">✕</span>`;
            container.appendChild(div);
        });
    }

    function newScheme() {
        const name = prompt('请输入新方案名称：', '方案' + (schemes.length+1));
        if (!name) return;
        const id = Date.now().toString(36) + Math.random().toString(36).substr(2,4);
        const scheme = {
            id: id,
            name: name,
            baseKey: '',
            keywords: []
        };
        schemes.push(scheme);
        saveSchemes();
        selectScheme(id);
        document.getElementById('schemeEditor').style.display = 'block';
    }

    function selectScheme(id) {
        currentSchemeId = id;
        const scheme = schemes.find(s => s.id === id);
        if (!scheme) {
            document.getElementById('schemeEditor').style.display = 'none';
            return;
        }
        document.getElementById('schemeEditor').style.display = 'block';
        document.getElementById('schemeName').value = scheme.name;
        renderKeywords(scheme);
        updateKeywordCount(scheme);
        document.getElementById('schemeSelector').value = id;
    }

    function syncKeywordsFromDOM() {
        const scheme = schemes.find(s => s.id === currentSchemeId);
        if (!scheme) return;
        const rows = document.querySelectorAll('.keyword-row');
        const keywords = [];
        rows.forEach(row => {
            const sheet = row.querySelector('.kw-sheet').value;
            const text = row.querySelector('.kw-text').value;
            const valuePosition = row.querySelector('.kw-position').value;
            const collectType = row.querySelector('.kw-collect-type').value;
            const offset = parseInt(row.querySelector('.kw-offset').value) || 1;
            const searchRange = row.querySelector('.kw-search-range').value;
            let startRow = null, endRow = null, rowNumber = null;
            if (searchRange === 'rows') {
                startRow = parseInt(row.querySelector('.kw-start-row').value) || null;
                endRow = parseInt(row.querySelector('.kw-end-row').value) || null;
            } else if (searchRange === 'row') {
                rowNumber = parseInt(row.querySelector('.kw-row-number').value) || null;
            }
            const excludeEmpty = row.querySelector('.kw-exclude-empty').checked;
            const excludeValues = row.querySelector('.kw-exclude-values').value.split(',').map(s => s.trim()).filter(s => s);
            const collectColor = row.querySelector('.kw-collect-color').checked;
            const addColorCol = row.querySelector('.kw-add-color').checked;
            if (sheet && text) {
                keywords.push({
                    sheet: sheet,
                    keywordText: text,
                    valuePosition: valuePosition,
                    collectType: collectType,
                    offset: offset,
                    searchRange: searchRange,
                    startRow: startRow,
                    endRow: endRow,
                    rowNumber: rowNumber,
                    excludeEmpty: excludeEmpty,
                    excludeValues: excludeValues,
                    collectColor: collectColor,
                    addColorCol: addColorCol
                });
            }
        });
        scheme.keywords = keywords;
    }

    function importSchemes(event) {
        const file = event.target.files[0];
        if (!file) return;
        const reader = new FileReader();
        reader.onload = (e) => {
            try {
                const imported = JSON.parse(e.target.result);
                if (!Array.isArray(imported)) throw new Error('格式错误');
                imported.forEach(s => {
                    const exist = schemes.findIndex(x => x.id === s.id);
                    if (exist >= 0) schemes[exist] = s;
                    else schemes.push(s);
                });
                saveSchemes();
                if (schemes.length > 0) selectScheme(schemes[0].id);
                alert('导入成功');
                log('导入 ' + imported.length + ' 个方案', 'success');
            } catch(err) {
                alert('导入失败：' + err.message);
                log('导入失败：' + err.message, 'error');
            }
        };
        reader.readAsText(file);
        event.target.value = '';
    }

    // ----- 文件上传 -----
    function handleFiles(filesOrEvent) {
        let files;
        if (filesOrEvent && filesOrEvent.target) {
            files = filesOrEvent.target.files;
        } else if (filesOrEvent && filesOrEvent.length !== undefined) {
            files = filesOrEvent;
        } else {
            return;
        }
        for (let f of files) {
            if (!f.name.endsWith('.xlsx') && !f.name.endsWith('.xls')) {
                alert(`文件 ${f.name} 不是 .xlsx 格式，跳过`);
                log(`文件 ${f.name} 格式不支持，跳过`, 'error');
                continue;
            }
            const reader = new FileReader();
            reader.onload = (e) => {
                const data = e.target.result;
                uploadedFiles.push({ name: f.name, data: data });
                renderFileList();
                log(`文件 ${f.name} 上传成功`, 'success');
            };
            reader.readAsArrayBuffer(f);
        }
        if (filesOrEvent && filesOrEvent.target) filesOrEvent.target.value = '';
    }

    // 统计执行（使用 ExcelJS 解析）

    async function runStatistics() {
        const scheme = schemes.find(s => s.id === currentSchemeId);
        if (!scheme) { alert('请先选择或创建方案'); return; }
        if (scheme.keywords.length === 0) { alert('方案中没有关键字'); return; }
        if (uploadedFiles.length === 0) { alert('请先上传Excel文件'); return; }

        log('开始统计...', 'info');
        const workbooks = [];
        let hasError = false;

        for (let f of uploadedFiles) {
            try {
                const wb = new ExcelJS.Workbook();
                await wb.xlsx.load(f.data);
                workbooks.push({ name: f.name, workbook: wb });
                log(`文件 ${f.name} 解析成功`, 'success');
            } catch(e) {
                const errorMsg = `文件 ${f.name} 解析失败：${e.message}\n堆栈：${e.stack}`;
                log(errorMsg, 'error');
                console.error(errorMsg);
                // 提示用户可能的解决方案
                if (e.message.includes('comments')) {
                    alert(`文件 ${f.name} 包含批注，ExcelJS 无法解析。请用 Excel 打开并删除所有批注（审阅→删除所有批注），然后重新上传。`);
                } else {
                    alert(`解析文件 ${f.name} 失败！\n请确认文件未损坏且为有效的 Excel 格式。\n详细错误请查看控制台或日志窗口。`);
                }
                hasError = true;
            }
        }

        if (workbooks.length === 0) {
            alert('没有可用的文件，请检查上传的文件是否有效');
            log('没有可用的文件', 'error');
            return;
        }

        const results = [];
        for (let wbData of workbooks) {
            try {
                const result = processWorkbookExcelJS(wbData.workbook, scheme);
                if (result) {
                    results.push({ name: wbData.name, data: result });
                    log(`文件 ${wbData.name} 统计完成`, 'success');
                } else {
                    hasError = true;
                }
            } catch(e) {
                log(`文件 ${wbData.name} 统计异常：${e.message}\n${e.stack}`, 'error');
                hasError = true;
            }
        }
        if (results.length === 0) {
            alert('所有文件统计均失败，请查看日志');
            log('所有文件统计失败', 'error');
            return;
        }
        statisticsResults = { scheme: scheme, results: results };
        displayResults(statisticsResults);
        document.getElementById('resultArea').style.display = 'block';
        document.getElementById('statusMsg').textContent = `统计完成，共 ${results.length} 个文件`;
        log(`统计完成，共 ${results.length} 个文件成功`, 'success');
        if (hasError) {
            log('部分文件存在错误，请查看日志', 'error');
        }
    }

    // ----- 核心数据处理（ExcelJS 版本）-----
    function processWorkbookExcelJS(workbook, scheme) {
        const keywords = scheme.keywords;
        const baseKeyIdx = scheme.baseKey !== '' ? parseInt(scheme.baseKey) : -1;

        const kwInfos = [];
        for (let idx = 0; idx < keywords.length; idx++) {
            const kw = keywords[idx];
            const sheet = workbook.getWorksheet(kw.sheet);
            if (!sheet) {
                log(`Sheet "${kw.sheet}" 未找到`, 'error');
                return null;
            }

            // 关键字搜索范围
            let startRow = 1;
            let endRow = sheet.rowCount;
            if (kw.searchRange === 'rows') {
                startRow = kw.startRow || 1;
                endRow = kw.endRow || sheet.rowCount;
            } else if (kw.searchRange === 'row') {
                startRow = kw.rowNumber || 1;
                endRow = startRow;
            }

            let col = null;
            let rowNum = null;
            for (let r = startRow; r <= endRow; r++) {
                const row = sheet.getRow(r);
                if (row.hidden) continue;
                let found = false;
                row.eachCell((cell, colNumber) => {
                    if (!col && cell.value && cell.value.toString().includes(kw.keywordText)) {
                        col = colNumber;
                        rowNum = r;
                        found = true;
                    }
                });
                if (found) break;
            }
            if (!col || !rowNum) {
                log(`在Sheet "${kw.sheet}" 的指定范围内未找到关键字 "${kw.keywordText}"`, 'error');
                return null;
            }

            // 值的采集位置
            let targetCol = col;
            if (kw.valuePosition === 'left') {
                targetCol = col - (kw.offset || 1);
            } else if (kw.valuePosition === 'right') {
                targetCol = col + (kw.offset || 1);
            }
            if (targetCol < 1) targetCol = 1;

            kwInfos.push({ sheet, col: targetCol, rowNum, kw, index: idx });
        }

        // ---- cell 模式 ----
        const cellData = [];
        for (let info of kwInfos) {
            if (info.kw.collectType !== 'cell') continue;
            const row = info.sheet.getRow(info.rowNum);
            const cell = row.getCell(info.col);
            let val = cell.value;
            if (val && typeof val === 'object' && val.result) val = val.result;
            else if (val && val.text) val = val.text;
            else if (val && val.richText) val = val.richText.map(t => t.text).join('');
            else if (val && typeof val === 'object') val = JSON.stringify(val);
            else if (val === undefined || val === null) val = null;

            let color = 'blank';
            if (info.kw.collectColor) {
                color = getCellColorName(cell);
            }

            let valid = true;
            if (info.kw.excludeEmpty && (val === null || val === undefined || val === '')) valid = false;
            if (valid && info.kw.excludeValues && info.kw.excludeValues.length > 0) {
                if (val !== null && val !== undefined) {
                    const strVal = String(val).trim();
                    if (info.kw.excludeValues.some(ev => ev.trim() === strVal)) valid = false;
                }
            }
            if (valid) {
                cellData.push({
                    keyword: info.kw,
                    value: val,
                    color: color
                });
            }
        }

        // ---- column 模式 ----
        const colInfos = kwInfos.filter(info => info.kw.collectType === 'column');
        const columnData = { rows: [], keywords: colInfos.map(info => info.kw) };
        if (colInfos.length > 0) {
            // 收集所有可见行（排除隐藏行和表头行）
            const allRowNumbers = new Set();
            colInfos.forEach(info => {
                const sheet = info.sheet;
                for (let r = 1; r <= sheet.rowCount; r++) {
                    const row = sheet.getRow(r);
                    if (row.hidden) continue;
                    if (r === info.rowNum) continue;
                    allRowNumbers.add(r);
                }
            });

            for (let rowNum of allRowNumbers) {
                const rowValues = [];
                let valid = true;
                for (let info of colInfos) {
                    const row = info.sheet.getRow(rowNum);
                    const cell = row.getCell(info.col);
                    let val = cell.value;
                    if (val && typeof val === 'object' && val.result) val = val.result;
                    else if (val && val.text) val = val.text;
                    else if (val && val.richText) val = val.richText.map(t => t.text).join('');
                    else if (val && typeof val === 'object') val = JSON.stringify(val);
                    else if (val === undefined || val === null) val = null;
                    let color = 'blank';
                    if (info.kw.collectColor) {
                        color = getCellColorName(cell);
                    }
                    rowValues.push({ value: val, color: color });
                }

                // 基准列非空检查
                if (baseKeyIdx >= 0) {
                    const colIndex = colInfos.findIndex(info => info.index === baseKeyIdx);
                    if (colIndex >= 0) {
                        const baseVal = rowValues[colIndex].value;
                        if (baseVal === null || baseVal === undefined || baseVal === '') {
                            continue;
                        }
                    }
                }

                // 排除空和特定值
                for (let i = 0; i < colInfos.length; i++) {
                    const kw = colInfos[i].kw;
                    const val = rowValues[i].value;
                    if (kw.excludeEmpty) {
                        if (val === null || val === undefined || val === '') {
                            valid = false;
                            break;
                        }
                    }
                    if (kw.excludeValues && kw.excludeValues.length > 0) {
                        if (val !== null && val !== undefined) {
                            const strVal = String(val).trim();
                            if (kw.excludeValues.some(ev => ev.trim() === strVal)) {
                                valid = false;
                                break;
                            }
                        }
                    }
                }
                if (valid) columnData.rows.push(rowValues);
            }
        }

        function getKeywordLabel(kw) {
            let label = kw.keywordText || ('关键字'+(kw.index+1));
            if (kw.valuePosition === 'left') label += ' ←' + kw.offset;
            else if (kw.valuePosition === 'right') label += ' →' + kw.offset;
            return label;
        }

        return {
            cellData: cellData,
            columnData: columnData,
            keywordLabels: keywords.map(k => getKeywordLabel(k)),
            cellKeywordLabels: keywords.filter(k => k.collectType==='cell').map(k => getKeywordLabel(k)),
            columnKeywordLabels: keywords.filter(k => k.collectType==='column').map(k => getKeywordLabel(k)),
            hasCell: cellData.length > 0,
            hasColumn: columnData.rows.length > 0
        };
    }

    function renderIndependentResults(stats) {
        const container = document.getElementById('independentResults');
        container.innerHTML = '';
        if (stats.results.length === 0) {
            container.innerHTML = '<p>无数据</p>';
            return;
        }

        if (independentLayout === 'horizontal') {
            const scrollDiv = document.createElement('div');
            scrollDiv.className = 'horizontal-scroll';
            stats.results.forEach((res, idx) => {
                const card = document.createElement('div');
                card.className = 'file-card';
                const title = document.createElement('h4');
                title.textContent = `📄 ${res.name}`;
                card.appendChild(title);
                const content = buildFileContent(res.data);
                card.appendChild(content);
                scrollDiv.appendChild(card);
            });
            container.appendChild(scrollDiv);
        } else {
            const tabsDiv = document.createElement('div');
            tabsDiv.className = 'file-tabs';
            const contentsDiv = document.createElement('div');
            contentsDiv.className = 'file-tab-contents';

            stats.results.forEach((res, idx) => {
                const tabBtn = document.createElement('button');
                tabBtn.className = 'tab-btn' + (idx === 0 ? ' active' : '');
                tabBtn.textContent = res.name;
                tabBtn.dataset.index = idx;
                tabBtn.onclick = function() {
                    document.querySelectorAll('.file-tabs .tab-btn').forEach(b => b.classList.remove('active'));
                    this.classList.add('active');
                    document.querySelectorAll('.file-tab-content').forEach(c => c.classList.remove('active'));
                    document.getElementById('file-tab-' + idx).classList.add('active');
                };
                tabsDiv.appendChild(tabBtn);

                const contentDiv = document.createElement('div');
                contentDiv.className = 'file-tab-content' + (idx === 0 ? ' active' : '');
                contentDiv.id = 'file-tab-' + idx;
                const title = document.createElement('h4');
                title.textContent = `📄 ${res.name}`;
                contentDiv.appendChild(title);
                const content = buildFileContent(res.data);
                contentDiv.appendChild(content);
                contentsDiv.appendChild(contentDiv);
            });
            container.appendChild(tabsDiv);
            container.appendChild(contentsDiv);
        }
    }

    function buildFileContent(data) {
        const frag = document.createElement('div');
        if (data.hasCell) {
            const statCard = document.createElement('div');
            statCard.className = 'stat-card';
            const title = document.createElement('div');
            title.className = 'stat-title';
            title.textContent = '单元格采集项';

            tableWrap.className = 'table-wrap';
            const table = buildCellResultTable(data.cellData);
            tableWrap.appendChild(table);

            title.textContent = '列采集项';

            tableWrap.appendChild(table);

        }
        return frag;
    }

    // ----- 表格构建函数（颜色列显示含义）-----
    function buildCellResultTable(cellData) {
        const table = document.createElement('table');
        const thead = document.createElement('thead');
        const trHead = document.createElement('tr');
        trHead.innerHTML = '<th>关键字</th><th>值</th>';
        let hasColor = cellData.some(d => d.keyword.collectColor);
        if (hasColor) trHead.innerHTML += '<th>结果</th>';
        thead.appendChild(trHead);
        table.appendChild(thead);
        const tbody = document.createElement('tbody');
        cellData.forEach(d => {
            const tr = document.createElement('tr');
            let label = d.keyword.keywordText;
            if (d.keyword.valuePosition === 'left') label += ' ←' + d.keyword.offset;
            else if (d.keyword.valuePosition === 'right') label += ' →' + d.keyword.offset;
            const tdLabel = document.createElement('td');
            tdLabel.textContent = label;
            tr.appendChild(tdLabel);
            const tdVal = document.createElement('td');
            tdVal.textContent = d.value !== null && d.value !== undefined ? d.value : '';
            if (d.value === null || d.value === undefined || d.value === '') tdVal.className = 'empty-value';
            if (d.keyword.collectColor && d.color && d.color !== 'blank') {
                tdVal.style.backgroundColor = getColorCss(d.color);
            }
            tr.appendChild(tdVal);
            if (hasColor) {
                const tdColor = document.createElement('td');
                const meaning = getColorMeaning(d.color);
                tdColor.textContent = meaning;
                const colorClass = 'color-' + (d.color === 'blank' ? 'blank' : d.color.toLowerCase());
                tdColor.className = colorClass;
                tr.appendChild(tdColor);
            }
            tbody.appendChild(tr);
        });
        table.appendChild(tbody);
        return table;
    }

    function buildMergedColumnTableMulti(fileResults) {
        const validFiles = fileResults.filter((res, idx) => res.data.hasColumn);
        if (validFiles.length === 0) {
            const p = document.createElement('p');
            p.textContent = '没有文件包含列数据';
            return p;
        }

        const firstKw = validFiles[0].data.columnData.keywords;
        const allSame = validFiles.every(r => {
            const kw = r.data.columnData.keywords;
            return kw.length === firstKw.length && kw.every((k, i) => k.keywordText === firstKw[i].keywordText);
        });
        if (!allSame) {
            const p = document.createElement('p');
            p.textContent = '各文件列模式关键字不一致，无法合并对比';
            return p;
        }

        const fileRows = validFiles.map((res, fi) => {
            const rows = res.data.columnData.rows;
            return rows.map(row => {
                const keyParts = row.map(cell => String(cell.value !== null && cell.value !== undefined ? cell.value : ''));
                const key = keyParts.join('|||');
                return { key, row, fileIndex: fi };
            });
        });

        const allKeys = new Set();
        fileRows.forEach(rows => rows.forEach(r => allKeys.add(r.key)));
        const sortedKeys = Array.from(allKeys).sort();

        const table = document.createElement('table');
        const thead = document.createElement('thead');
        const trHead = document.createElement('tr');
        trHead.innerHTML = '<th>来源文件</th>';
        const hasColor = firstKw.some(kw => kw.collectColor);
        firstKw.forEach(kw => {
            let label = kw.keywordText;
            if (kw.valuePosition === 'left') label += ' ←' + kw.offset;
            else if (kw.valuePosition === 'right') label += ' →' + kw.offset;
            trHead.innerHTML += `<th>${label}</th>`;
            if (kw.addColorCol) {
                trHead.innerHTML += `<th>结果</th>`;
            }
        });
        thead.appendChild(trHead);
        table.appendChild(thead);

        const tbody = document.createElement('tbody');
        const bgColors = ['#e6f3e6', '#fff9e6', '#f0ede6', '#f0e6f0', '#e6f0f0', '#f0e6e6'];

        sortedKeys.forEach(key => {
            const rowsWithKey = [];
            validFiles.forEach((res, fi) => {
                const rows = fileRows[fi];
                const found = rows.find(r => r.key === key);
                if (found) {
                    rowsWithKey.push({ fileIndex: fi, row: found.row });
                }
            });
            rowsWithKey.sort((a, b) => a.fileIndex - b.fileIndex);
            rowsWithKey.forEach(({ fileIndex, row }) => {
                const tr = document.createElement('tr');
                tr.style.backgroundColor = bgColors[fileIndex % bgColors.length];
                const tdFile = document.createElement('td');
                tdFile.textContent = validFiles[fileIndex].name;
                tr.appendChild(tdFile);
                row.forEach((cell, idx) => {
                    const kw = firstKw[idx];
                    const td = document.createElement('td');
                    td.textContent = cell.value !== null && cell.value !== undefined ? cell.value : '';
                    if (cell.value === null || cell.value === undefined || cell.value === '') td.className = 'empty-value';
                    if (kw.collectColor && cell.color && cell.color !== 'blank') {
                        td.style.backgroundColor = getColorCss(cell.color);
                    }
                    tr.appendChild(td);
                    if (kw.addColorCol) {
                        const tdColor = document.createElement('td');
                        const meaning = getColorMeaning(cell.color);
                        tdColor.textContent = meaning;
                        const colorClass = 'color-' + (cell.color === 'blank' ? 'blank' : cell.color.toLowerCase());
                        tdColor.className = colorClass;
                        tr.appendChild(tdColor);
                    }
                });
                tbody.appendChild(tr);
            });
        });

        table.appendChild(tbody);
        return table;
    }

    async function exportIndependentExcel() {
        if (!statisticsResults) return alert('请先执行统计');
        const wb = new ExcelJS.Workbook();
        statisticsResults.results.forEach((res, fi) => {
            const ws = wb.addWorksheet(res.name || ('文件'+(fi+1)));
            let rowIdx = 1;
            if (res.data.hasCell) {
                ws.addRow(['单元格采集项']);
                const headers = ['关键字', '值'];
                const hasColor = res.data.cellData.some(d => d.keyword.collectColor);
                if (hasColor) headers.push('结果');
                const headerRow = ws.addRow(headers);
                headerRow.font = { bold: true };
                res.data.cellData.forEach(d => {
                    const row = [];
                    let label = d.keyword.keywordText;
                    if (d.keyword.valuePosition === 'left') label += ' ←' + d.keyword.offset;
                    else if (d.keyword.valuePosition === 'right') label += ' →' + d.keyword.offset;
                    row.push(label);
                    row.push(d.value !== null && d.value !== undefined ? d.value : '');
                    if (hasColor) {
                        const meaning = getColorMeaning(d.color);
                        row.push(meaning);
                    }
                    const rowObj = ws.addRow(row);
                    if (d.keyword.collectColor && d.color && d.color !== 'blank') {
                        const cell = rowObj.getCell(2);
                        const argb = getColorArgb(d.color);
                        if (argb) cell.fill = { type: 'pattern', pattern: 'solid', fgColor: { argb: argb } };
                    }
                });
                rowIdx = ws.rowCount + 2;
            }
            if (res.data.hasColumn) {
                if (rowIdx > 1) ws.addRow([]);
                ws.addRow(['列采集项']);
                const colData = res.data.columnData;
                const headers = [];
                const colorColIndices = [];
                colData.keywords.forEach((kw, idx) => {
                    let label = kw.keywordText;
                    if (kw.valuePosition === 'left') label += ' ←' + kw.offset;
                    else if (kw.valuePosition === 'right') label += ' →' + kw.offset;
                    headers.push(label);
                    if (kw.addColorCol) {
                        headers.push('结果');
                        colorColIndices.push(idx * 2 + 1);
                    }
                });
                const headerRow = ws.addRow(headers);
                headerRow.font = { bold: true };
                colData.rows.forEach(row => {
                    const values = [];
                    row.forEach((cell, idx) => {
                        values.push(cell.value !== null && cell.value !== undefined ? cell.value : '');
                        if (colData.keywords[idx].addColorCol) {
                            const meaning = getColorMeaning(cell.color);
                            values.push(meaning);
                        }
                    });
                    const rowObj = ws.addRow(values);
                    row.forEach((cell, idx) => {
                        const kw = colData.keywords[idx];
                        if (kw.collectColor && cell.color && cell.color !== 'blank') {
                            const cellExcel = rowObj.getCell(idx + 1);
                            const argb = getColorArgb(cell.color);
                            if (argb) cellExcel.fill = { type: 'pattern', pattern: 'solid', fgColor: { argb: argb } };
                        }
                    });
                    colorColIndices.forEach(colIndex => {
                        const cell = rowObj.getCell(colIndex + 1);
                        const colorName = values[colIndex];
                        const argb = getColorArgb(colorName);
                        if (argb) {
                            cell.fill = { type: 'pattern', pattern: 'solid', fgColor: { argb: argb } };
                        }
                    });
                });
            }
            ws.columns.forEach(col => col.width = 15);
        });
        const buf = await wb.xlsx.writeBuffer();
        saveAs(new Blob([buf], { type: 'application/octet-stream' }), '独立统计结果.xlsx');
        log('独立统计 Excel 导出成功', 'success');
    }
