<h3>粘贴 Excel 三列数据（含换行/引号也兼容）</h3>

<button onclick="buildScript()">生成并复制脚本</button>
<button onclick="clearAll()">清空</button>

<div class="tipOk" id="tipOk">✅ 生成完成并已复制！</div>

<script>
function escapeAll(str){
    return str
        .replace(/\\/g, '\\\\')
        .replace(/`/g, '\\`')
        .replace(/"/g, '\\"')
        .replace(/\r/g, '\\r')
        .replace(/\n/g, '\\n');
}

function buildScript(){
    document.getElementById('tipOk').style.display = 'none';
    document.getElementById('tipErr').style.display = 'none';

    var raw = document.getElementById('inputArea').value;
    if(!raw.trim()){
        document.getElementById('tipErr').style.display = 'block';
        return;
    }

    var safeData = escapeAll(raw);

    var js = '(function(){\n'
+'const excelDataRaw = `' + safeData + '`;\n'
+'try{\n'
+'    const dataMap = {};\n'
+'    excelDataRaw.split("\\n").forEach(line=>{\n'
+'        const row = line.trim();\n'
+'        if(!row) return;\n'
+'        const cells = row.split(/\\t+/);\n'
+'        const key = (cells[0]||"").trim();\n'
+'        const col2 = (cells[1]||"").trim();\n'
+'        const col3 = (cells.slice(2).join("\\t")||"").trim();\n'
+'        if(key) dataMap[key] = {col2, col3};\n'
+'    });\n'
+'    if(Object.keys(dataMap).length===0) return alert("无有效数据");\n'
+'\n'
+'    const config = {\n'
+'        rowSelector: "table tr:has(td)",\n'
+'        style: "background:#e8f4ff;padding:4px 8px;border:1px solid #409eff;color:#333;cursor:pointer;",\n'
+'        cls: "excel-matched-col"\n'
+'    };\n'
+'    const processed = new WeakSet();\n'
+'\n'
+'    function renderTable(){\n'
+'        const rows = document.querySelectorAll(config.rowSelector);\n'
+'        rows.forEach(tr=>{\n'
+'            if(processed.has(tr)) return;\n'
+'            const tds = tr.querySelectorAll("td");\n'
+'            let key = "", keyTd = null;\n'
+'            for(let td of tds){\n'
+'                const txt = td.textContent.trim();\n'
+'                if(dataMap[txt]){ key=txt; keyTd=td; break; }\n'
+'            }\n'
+'            if(!keyTd) return;\n'
+'\n'
+'            const d = dataMap[key];\n'
+'            const td2 = document.createElement("td");\n'
+'            td2.className = config.cls;\n'
+'            td2.style.cssText = config.style;\n'
+'            td2.textContent = d.col2;\n'
+'            td2.title = "点击复制";\n'
+'            td2.onclick = function(){\n'
+'                const t = document.createElement("textarea");\n'
+'                t.value = this.textContent;\n'
+'                document.body.appendChild(t);\n'
+'                t.select();\n'
+'                document.execCommand("copy");\n'
+'                document.body.removeChild(t);\n'
+'                alert("已复制："+this.textContent);\n'
+'            };\n'
+'\n'
+'            const td3 = document.createElement("td");\n'
+'            td3.className = config.cls;\n'
+'            td3.style.cssText = config.style;\n'
+'            td3.textContent = d.col3;\n'
+'            td3.title = "点击复制";\n'
+'            td3.onclick = function(){\n'
+'                const t = document.createElement("textarea");\n'
+'                t.value = this.textContent;\n'
+'                document.body.appendChild(t);\n'
+'                t.select();\n'
+'                document.execCommand("copy");\n'
+'                document.body.removeChild(t);\n'
+'                alert("已复制："+this.textContent);\n'
+'            };\n'
+'\n'
+'            keyTd.after(td2, td3);\n'
+'            processed.add(tr);\n'
+'        });\n'
+'    }\n'
+'\n'
+'    renderTable();\n'
+'    let timer = null;\n'
+'    const ob = new MutationObserver(()=>{\n'
+'        clearTimeout(timer);\n'
+'        timer = setTimeout(renderTable,200);\n'
+'    });\n'
+'    ob.observe(document.body, {childList:true, subtree:true});\n'
+'    console.log("✅ 表格填充脚本已启动，共"+Object.keys(dataMap).length+"条数据");\n'
+'}catch(e){\n'
+'    alert("脚本异常："+e.message);\n'
+'    console.error(e);\n'
+'}\n'
+'})();';

    var out = document.getElementById('outArea');
    out.value = js;

    // 复制
    out.select();
    document.execCommand('copy');
    document.getElementById('tipOk').style.display = 'block';
}

function clearAll(){
    document.getElementById('inputArea').value = '';
    document.getElementById('outArea').value = '';
    document.getElementById('tipOk').style.display = 'none';
    document.getElementById('tipErr').style.display = 'none';
}
</script>
