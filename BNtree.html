from bs4 import BeautifulSoup from lxml import etree import os import tkinter as tk from tkinter import filedialog def get_full_xpath(element): path_parts = [] current = element while current is not None and current.tag is not None: parent = current.getparent() if parent is None: break siblings = [s for s in parent.iterchildren(tag=current.tag)] idx = siblings.index(current) + 1 path_parts.insert(0, f"{current.tag}[{idx}]") current = parent return "/" + "/".join(path_parts) if path_parts else "" def add_xpath_to_html(html_path, save_path): with open(html_path, "r", encoding="utf-8", errors="ignore") as f: html_content = f.read() parser = etree.HTMLParser() tree = etree.fromstring(html_content, parser) soup = BeautifulSoup(html_content, "html.parser") all_elements = tree.iter() for elem in all_elements: if elem.text and elem.text.strip(): xpath = get_full_xpath(elem) new_text = f"{elem.text.strip()} 【XPath: {xpath}】" try: bs_elem = soup.find(elem.tag, string=lambda t: t and elem.text.strip() in t) if bs_elem: bs_elem.string = new_text except: continue new_html = soup.prettify(encoding="utf-8") with open(save_path, "wb") as f: f.write(new_html) return True def main(): root = tk.Tk() root.withdraw() root.title("HTML XPath追加工具") print("请在弹窗中选择要处理的HTML文件...") file_path = filedialog.askopenfilename( title="选择HTML文件", filetypes=[("HTML 文件", "*.html;*.htm"), ("所有文件", "*.*")] ) if not file_path: print("未选择文件，退出。") return dir_name, file_name = os.path.split(file_path) name, ext = os.path.splitext(file_name) out_file = os.path.join(dir_name, f"{name}_with_xpath{ext}") try: add_xpath_to_html(file_path, out_file) print(f"\n✅ 处理成功！\n文件已保存：{out_file}") except Exception as e: print(f"\n❌ 处理失败：{str(e)}") input("\n按下回车关闭窗口...") if __name__ == "__main__": main()
